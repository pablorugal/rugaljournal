function SectionLabel({ children }: { children: React.ReactNode }) {
  return (
    <div className="flex items-center gap-2 px-3 mb-2">
      <p className="font-mono text-[10px] font-medium uppercase tracking-[0.16em] text-ink-900/35 dark:text-bone-100/35">{children}</p>
      <span className="h-px flex-1 bg-gradient-to-r from-black/10 dark:from-brand/25 to-transparent" />
    </div>
  )
}

function NavItem({ to, label, icon: Icon, end, badge }: { to: string; label: string; icon: any; end?: boolean; badge?: string }) {
  return (
    <NavLink to={to} end={end} className={({ isActive }) =>
      `tj-nav-item group flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm font-medium transition ${isActive ? 'tj-nav-active bg-black/5 dark:bg-transparent text-ink-900 dark:text-brand font-semibold' : 'text-ink-900/60 dark:text-bone-100/60 hover:bg-black/5 dark:hover:bg-white/[0.04] hover:text-ink-900 dark:hover:text-bone-100'}`}>
      <Icon size={18} strokeWidth={2} className="tj-nav-icon transition duration-200 group-hover:scale-110 dark:group-hover:text-brand" />
      <span>{label}</span>
      {badge && (
        <span className="ml-auto font-mono text-[9px] font-semibold tracking-[0.14em] px-1.5 py-0.5 rounded-md border border-brand-deep/40 dark:border-brand/40 text-brand-deep dark:text-brand bg-brand/10">{badge}</span>
      )}
    </NavLink>
  )
}
