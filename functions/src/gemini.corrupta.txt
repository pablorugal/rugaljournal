import {onCall, HttpsError} from "firebase-functions/v2/https";
import {defineSecret} from "firebase-functions/params";
import * as logger from "firebase-functions/logger";
import {consumeGeminiQuota} from "./rateLimit";

// La clave vive en Secret Manager. Nunca va en el código ni en el navegador.
export const GEMINI_API_KEY = defineSecret("GEMINI_API_KEY");

// Misma lista que tenías en el navegador.
const MODEL_FALLBACK_LIST = [
  "gemini-3.8-flash",
  "gemini-3.5-flash-lite",
  "gemini-flash-latest",
  "gemini-flash-lite-latest",
];

const RETRIES_PER_MODEL = 2; // reintentos ante 503 antes de pasar al siguiente modelo
const RETRY_DELAY_MS = 1500;
const REQUEST_TIMEOUT_MS = 40000; // tope por cada llamada a Gemini
const TOTAL_BUDGET_MS = 100000; // tope total por defecto (la función tiene 120 s)

interface GeminiApiResponse {
  error?: {message?: string};
  candidates?: Array<{
    finishReason?: string;
    content?: {parts?: Array<{text?: string}>};
  }>;
}

type SingleResult =
  | {ok: true; text: string}
  | {ok: false; status: number | null; message: string};

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

async function callSingleModel(
  model: string,
  apiKey: string,
  body: object,
  timeoutMs: number,
  requireComplete: boolean
): Promise<SingleResult> {
  try {
    const response = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`,
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "x-goog-api-key": apiKey,
        },
        body: JSON.stringify(body),
        signal: AbortSignal.timeout(timeoutMs),
      }
    );

    let data: GeminiApiResponse;
    try {
      data = (await response.json()) as GeminiApiResponse;
    } catch {
      return {
        ok: false,
        status: response.status,
        message: `Respuesta no válida de Gemini (HTTP ${response.status})`,
      };
    }

    if (data.error) {
      return {
        ok: false,
        status: response.status,
        message: data.error.message || "Error desconocido",
      };
    }

    const candidate = data.candidates?.[0];
    const text = candidate?.content?.parts?.[0]?.text;
    if (text) {
      // Si quien llama exige respuesta completa, una respuesta cortada cuenta como fallo.
      if (requireComplete && candidate?.finishReason && candidate.finishReason !== "STOP") {
        return {
          ok: false,
          status: response.status,
          message: `Respuesta incompleta (finishReason ${candidate.finishReason})`,
        };
      }
      return {ok: true, text};
    }

    return {ok: false, status: response.status, message: "Respuesta vacía del modelo"};
  } catch (err) {
    const message = err instanceof Error ? err.message : "Error de red";
    return {ok: false, status: null, message};
  }
}

function allFailed(errors: string[]): HttpsError {
  return new HttpsError(
    "unavailable",
    "Todos los modelos gratuitos fallaron (probablemente alta demanda temporal en Google). " +
      `Detalle: ${errors.join(" | ")}`
  );
}

export async function generateWithFallback(
  apiKey: string,
  body: object,
  options: {totalBudgetMs?: number; requireComplete?: boolean} = {}
): Promise<{text: string; model: string}> {
  const budget = options.totalBudgetMs ?? TOTAL_BUDGET_MS;
  const requireComplete = options.requireComplete ?? false;
  const started = Date.now();
  const errors: string[] = [];

  for (const model of MODEL_FALLBACK_LIST) {
    let attempt = 0;

    while (attempt <= RETRIES_PER_MODEL) {
      const remaining = budget - (Date.now() - started);
      if (remaining < 5000) {
        errors.push("tiempo máximo agotado");
        throw allFailed(errors);
      }

      const result = await callSingleModel(
        model,
        apiKey,
        body,
        Math.min(REQUEST_TIMEOUT_MS, remaining),
        requireComplete
      );

      if (result.ok) {
        return {text: result.text, model};
      }

      errors.push(`${model} [${result.status ?? "network"}]: ${result.message}`);
      logger.warn(
        `Modelo ${model} falló (HTTP ${result.status}, intento ${attempt + 1}): ${result.message}`
      );

      // Solo reintentamos ante 503 (sobrecarga temporal). 404/429/403 no mejoran reintentando.
      if (result.status !== 503) break;

      attempt++;
      if (attempt <= RETRIES_PER_MODEL) {
        await sleep(RETRY_DELAY_MS);
      }
    }
  }

  throw allFailed(errors);
}

/**
 * Proxy de Gemini. El navegador envía el cuerpo de la petición y el servidor
 * añade la clave. Exige usuario autenticado y aplica un tope por usuario.
 */
export const callGemini = onCall(
  {
    region: "europe-west1",
    secrets: [GEMINI_API_KEY],
    timeoutSeconds: 120,
    maxInstances: 5,
  },
  async (request) => {
    const uid = request.auth?.uid;
    if (!uid) {
      throw new HttpsError("unauthenticated", "Debes iniciar sesión para usar Nova.");
    }

    const data = request.data as {body?: unknown} | undefined;
    const rawBody = data?.body;
    if (!rawBody || typeof rawBody !== "object" || Array.isArray(rawBody)) {
      throw new HttpsError("invalid-argument", "Petición no válida.");
    }

    const raw = rawBody as Record<string, unknown>;
    if (!Array.isArray(raw.contents) || raw.contents.length === 0) {
      throw new HttpsError("invalid-argument", "Falta el contenido de la petición.");
    }

    // Solo se reenvían estos tres campos. Cualquier otro se descarta.
    const safeBody: Record<string, unknown> = {contents: raw.contents};
    if (raw.systemInstruction) safeBody.systemInstruction = raw.systemInstruction;
    if (raw.generationConfig) safeBody.generationConfig = raw.generationConfig;

    // Tope por usuario: se descuenta antes de llamar a Gemini.
    await consumeGeminiQuota(uid);

    const {text, model} = await generateWithFallback(GEMINI_API_KEY.value(), safeBody);
    logger.info(`Gemini OK. uid=${uid} modelo=${model}`);
    return {text, model};
  }
);
